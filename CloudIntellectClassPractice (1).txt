    //write down method that queries 10 opportunity record create set of account id from it ,use set of Id to query list of opportunity ,return this list of opportuinity
public List<Opportunity> getOpportunities() {

    // Step 1: Query 10 Opportunity records
    List<Opportunity> oppList = [
        SELECT Id, Name, AccountId
        FROM Opportunity
        LIMIT 10
    ];

    // Step 2: Create a Set of Account IDs
    Set<Id> accountIds = new Set<Id>();

    for (Opportunity opp : oppList) {
        if (opp.AccountId != null) {
            accountIds.add(opp.AccountId);
        }
    }

    // Step 3: Query Opportunities related to those Accounts
    List<Opportunity> finalOppList = [
        SELECT Id, Name, AccountId
        FROM Opportunity
        WHERE AccountId IN :accountIds
    ];

    // Step 4: Return the Opportunity list
    return finalOppList;
}
