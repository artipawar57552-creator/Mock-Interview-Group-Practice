💡 Bonus Task (Optional): Contacts of Queried Accounts
Write an Apex program that:
• Queries 3 Accounts with their Ids.
• Uses those Ids in a second query to fetch related Contacts (WHERE AccountId IN
:accountList).
• Prints Account Name and Contact Names together.


// Step 1: Query 3 Accounts
List<Account> accountList = [
    SELECT Id, Name
    FROM Account
    LIMIT 3
];
// Step 2: Collect Account Ids
Set<Id> accountIds = new Set<Id>();

for (Account acc : accountList) {
    accountIds.add(acc.Id);
}

// Step 3: Query Contacts using Account Ids
List<Contact> contactList = [
    SELECT Id, FirstName, LastName, AccountId
    FROM Contact
    WHERE AccountId IN :accountIds     //con1,con2 ,con3
];

// Step 4: Print Account Name and related Contact Names
for (Account acc : accountList) {

    System.debug('Account: ' + acc.Name);

    for (Contact con : contactList) {

        if (con.AccountId == acc.Id) {
            System.debug(
                '   Contact: ' +
                con.FirstName + ' ' +
                con.LastName
            );
        }
    }
}
